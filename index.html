function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (sheet.getLastRow() === 0) { sheet.appendRow(["Timestamp","Date","Staff","Role","Areas Completed","Areas Skipped","Total Areas","Completed Count","Result","Notes"]); sheet.getRange(1,1,1,10).setBackground("#2d6a4f").setFontColor("#ffffff").setFontWeight("bold"); } var data = JSON.parse(e.postData.contents); sheet.appendRow([ data.timestamp || "", data.dateKey || "", data.staff || "", data.role || "", data.areasCompleted || "", data.areasSkipped || "", data.totalAreas || "", data.completedCount || "", data.result || "", data.notes || "" ]); var resultCell = sheet.getRange(sheet.getLastRow(), 9); if (data.result === "Pass") resultCell.setBackground("#d1fae5").setFontColor("#065f46"); else if (data.result === "Needs Attention") resultCell.setBackground("#fef3c7").setFontColor("#92400e"); else if (data.result === "Fail") resultCell.setBackground("#fee2e2").setFontColor("#991b1b"); return ContentService.createTextOutput(JSON.stringify({status:"ok"})).setMimeType(ContentService.MimeType.JSON); } function doGet(e) { return ContentService.createTextOutput("Cleaning checklist endpoint is live").setMimeType(ContentService.MimeType.TEXT); }
